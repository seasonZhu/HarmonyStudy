### 2.2 签名算法设计

```typescript
/**
 * 签名参数
 */
interface SignParams {
  timestamp: number;       // 时间戳 (毫秒)
  nonce: string;          // 随机数
  appId: string;          // 应用ID
  signType: string;       // 签名类型: MD5/SHA256
}

/**
 * 签名结果
 */
interface SignResult {
  signature: string;       // 签名
  timestamp: string;       // 时间戳字符串
  nonce: string;           // 随机数
  appId: string;           // 应用ID
}

/**
 * 签名算法
 *
 * 签名公式: MD5(appId + timestamp + nonce + secretKey)
 */
function signRequest(params: Record<string, string | number>, secretKey: string): SignResult {
  // 1. 生成时间戳和随机数
  const timestamp = Date.now().toString();
  const nonce = generateNonce();

  // 2. 拼接参数
  const sortedKeys = Object.keys(params).sort();
  const paramString = sortedKeys
    .map(key => `${key}=${params[key]}`)
    .join('&');

  // 3. 计算签名
  const signString = `appId=${APP_ID}&timestamp=${timestamp}&nonce=${nonce}&${paramString}&secretKey=${secretKey}`;
  const signature = CryptoUtil.sha256(signString);

  return {
    signature,
    timestamp,
    nonce,
    appId: APP_ID
  };
}

/**
 * 生成随机数
 */
function generateNonce(): string {
  return Math.random().toString(36).substring(2) + Date.now().toString(36);
}
```

## 3. 安全拦截器设计

### 3.1 签名拦截器

```typescript
import { AxiosClientRequestInterceptor } from './interceptors/AxiosClientRequestInterceptor';
import { CryptoUtil } from '../utils/CryptoUtil';

// 应用配置
const APP_ID = 'your_app_id';
const SECRET_KEY = 'your_secret_key';

/**
 * 签名拦截器
 *
 * 在每个请求中添加签名相关 headers
 */
class SignInterceptor implements AxiosClientRequestInterceptor {
  onFulfilled = (config: AxiosRequestConfig): AxiosRequestConfig => {
    // 添加时间戳
    const timestamp = Date.now().toString();
    // 添加随机数
    const nonce = CryptoUtil.generateUUID();
    // 添加 AppId
    const appId = APP_ID;

    // 设置 headers
    if (!config.headers) {
      config.headers = {};
    }

    config.headers['X-App-Id'] = appId;
    config.headers['X-Timestamp'] = timestamp;
    config.headers['X-Nonce'] = nonce;

    // 计算签名
    const signString = `${appId}${timestamp}${nonce}${SECRET_KEY}`;
    const signature = CryptoUtil.sha256(signString);
    config.headers['X-Signature'] = signature;

    return config;
  };

  onRejected = (error: Error): Error => {
    console.error('签名拦截器错误: ' + error.message);
    return error;
  };
}
```

### 3.2 敏感数据脱敏拦截器

```typescript
/**
 * 响应数据脱敏拦截器
 */
class DesensitizeInterceptor implements AxiosClientResponseInterceptor {
  onFulfilled = (response: AxiosResponse): AxiosResponse => {
    const data = response.data;

    if (data && typeof data === 'object') {
      // 脱敏处理
      data.phoneNumber = this.desensitize(data.phoneNumber, 3, 4);
      data.idCard = this.desensitize(data.idCard, 6, 4);
      data.email = this.desensitizeEmail(data.email);
    }

    return response;
  };

  /**
   * 手机号脱敏
   */
  private desensitize(value: string, prefixLen: number, suffixLen: number): string {
    if (!value || value.length < prefixLen + suffixLen) {
      return value;
    }
    const prefix = value.substring(0, prefixLen);
    const suffix = value.substring(value.length - suffixLen);
    const mask = '*'.repeat(value.length - prefixLen - suffixLen);
    return prefix + mask + suffix;
  }

  /**
   * 邮箱脱敏
   */
  private desensitizeEmail(email: string): string {
    if (!email || !email.includes('@')) {
      return email;
    }
    const [username, domain] = email.split('@');
    if (username.length <= 2) {
      return username[0] + '***@' + domain;
    }
    return username[0] + '***' + username[username.length - 1] + '@' + domain;
  }
}
```

## 4. 路由安全

### 4.1 路由守卫

```typescript
/**
 * 路由守卫
 *
 * 在路由跳转前进行权限校验
 */
class RouterGuard {
  // 需要登录的页面
  private static readonly LOGIN_REQUIRED_PATHS = [
    '/pages/My',
    '/pages/MyCoin',
    '/pages/MyCollect'
  ];

  /**
   * 路由前置校验
   */
  static beforeRoute(path: string, isLoggedIn: boolean): boolean {
    // 检查是否需要登录
    if (this.LOGIN_REQUIRED_PATHS.includes(path)) {
      if (!isLoggedIn) {
        // 未登录，跳转到登录页
        Router.push('/pages/Login');
        return false;
      }
    }
    return true;
  }
}
```

## 5. 安全编码规范

### 5.1 输入校验

```typescript
/**
 * 输入校验工具
 */
class InputValidator {
  /**
   * 校验手机号
   */
  static isValidPhone(phone: string): boolean {
    return /^1[3-9]\d{9}$/.test(phone);
  }

  /**
   * 校验邮箱
   */
  static isValidEmail(email: string): boolean {
    return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email);
  }

  /**
   * 校验 URL
   */
  static isValidUrl(url: string): boolean {
    try {
      new URL(url);
      return true;
    } catch {
      return false;
    }
  }

  /**
   * XSS 过滤
   */
  static escapeHtml(text: string): string {
    const map: Record<string, string> = {
      '&': '&amp;',
      '<': '&lt;',
      '>': '&gt;',
      '"': '&quot;',
      "'": '&#39;'
    };
    return text.replace(/[&<>"']/g, char => map[char]);
  }
}
```

### 5.2 敏感日志处理

```typescript
/**
 * 日志脱敏
 */
class SafeLogger {
  // 敏感字段名
  private static readonly SENSITIVE_FIELDS = [
    'password', 'token', 'secretKey', 'cookie', 'idCard', 'phoneNumber'
  ];

  /**
   * 日志脱敏处理
   */
  static sanitize(data: Record<string, any>): Record<string, any> {
    const sanitized = { ...data };

    for (const field of this.SENSITIVE_FIELDS) {
      if (sanitized[field]) {
        sanitized[field] = '***';
      }
    }

    return sanitized;
  }

  /**
   * 安全日志输出
   */
  static info(message: string, data?: Record<string, any>): void {
    if (data) {
      console.info(message, JSON.stringify(this.sanitize(data)));
    } else {
      console.info(message);
    }
  }
}
```
