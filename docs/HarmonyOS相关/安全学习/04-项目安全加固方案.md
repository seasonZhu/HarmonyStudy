```typescript
  /**
   * 安全存储 Token
   */
  static async saveTokenSecure(token: string): Promise<void> {
    // 使用 AES 加密
    const encrypted = await CryptoUtil.aesEncrypt(token, this.getMasterKey());

    // 存储加密后的 Token
    await persistenceStorage.put(this.ENCRYPTED_TOKEN_KEY, encrypted);
  }

  /**
   * 安全读取 Token
   */
  static async getTokenSecure(): Promise<string | null> {
    const encrypted = await persistenceStorage.get(this.ENCRYPTED_TOKEN_KEY) as string;
    if (!encrypted) return null;

    return await CryptoUtil.aesDecrypt(encrypted, this.getMasterKey());
```

#### 密码加强加密

```typescript
// PasswordCrypto 密码加密工具
class PasswordCrypto {
  private static readonly SALT_LENGTH = 16;
  private static readonly ITERATIONS = 10000;

  /**
   * 密码哈希（加盐迭代哈希）
   */
  static async hash(password: string, salt?: string): Promise<{ hash: string; salt: string }> {
    const usedSalt = salt || this.generateSalt();
    const combined = password + usedSalt;

    // 多次迭代哈希
    let hash = combined;
    for (let i = 0; i < ITERATIONS; i++) {
      hash = CryptoUtil.sha256(hash);
    }

    return {
      hash: hash,
      salt: usedSalt
    };
  }

  /**
   * 验证密码
   */
  static async verify(password: string, hash: string, salt: string): Promise<boolean> {
    const result = await this.hash(password, salt);
    return result.hash === hash;
  }

  private static generateSalt(): string {
    const random = new Uint8Array(SALT_LENGTH);
    for (let i = 0; i < SALT_LENGTH; i++) {
      random[i] = Math.floor(Math.random() * 256);
    }
    return CryptoUtil.base64Encode(random);
  }
}
```

#### 安全审计日志

```typescript
// 安全审计日志
class SecurityAuditLogger {
  private static readonly AUDIT_EVENTS = {
    LOGIN_SUCCESS: 'LOGIN_SUCCESS',
    LOGIN_FAILED: 'LOGIN_FAILED',
    TOKEN_REFRESH: 'TOKEN_REFRESH',
    PERMISSION_DENIED: 'PERMISSION_DENIED',
    SUSPICIOUS_ACTIVITY: 'SUSPICIOUS_ACTIVITY'
  };

  /**
   * 记录安全事件
   */
  static log(event: string, details: Record<string, any>): void {
    const auditEntry = {
      event: event,
      timestamp: Date.now(),
      deviceId: this.getDeviceId(),
      details: this.sanitize(details)
    };

    // 发送到服务器（异步，不阻塞主流程）
    this.sendToServer(auditEntry).catch(() => {
      // 本地缓存，待重试
      this.cacheAudit(auditEntry);
    });
  }

  /**
   * 脱敏敏感数据
   */
  private static sanitize(data: Record<string, any>): Record<string, any> {
    const sanitized = { ...data };
    const sensitiveKeys = ['password', 'token', 'secret', 'cookie'];

    for (const key of sensitiveKeys) {
      if (sanitized[key]) {
        sanitized[key] = '***';
      }
    }

    return sanitized;
  }
}
```

## 3. 安全检查清单

### 3.1 开发规范

- [ ] 所有 API 请求使用 HTTPS
- [ ] 敏感数据（Token、密码）不得明文存储

```
├─────────────────────────────────────────────────────────────┤
│                                                             │
│  Level 1: 主密钥 (Master Key)                              │

│                                                             │
│  Level 2: 数据密钥 (Data Key)                              │
│  └── 由主密钥加密后存储                                      │
│      └── 用于加密业务数据                                   │
│                                                             │
│  Level 3: 传输密钥 (Session Key)                           │
│  └── 每次会话生成                                           │
│      └── 用于单次请求加密                                   │
│                                                             │
└─────────────────────────────────────────────────────────────┘
```
