```typescript
// 缓存策略类型
enum CacheStrategy {
  NO_CACHE,        // 不缓存
  MEMORY_ONLY,     // 仅内存缓存
  DISK_ONLY,       // 仅磁盘缓存
  CACHE_FIRST,     // 缓存优先
  NETWORK_FIRST,   // 网络优先
  STALE_WHILE_REVALIDATE  // 后台更新
}

// 缓存配置
interface CacheConfig {
  strategy: CacheStrategy
  maxAge: number        // 缓存有效期（秒）
  staleWhileRevalidate: number  // 后台更新延迟
}
```
